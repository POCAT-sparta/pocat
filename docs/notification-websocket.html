// 클라이언트 연결 예시 (SockJS + stomp.js)
const socket = new SockJS('/ws');
const stompClient = Stomp.over(socket);

stompClient.connect(
  { Authorization: `Bearer ${token}` },  // CONNECT 헤더에 JWT
  () => {
    stompClient.subscribe(`/sub/notifications/${userId}`, (message) => {
      const data = JSON.parse(message.body);
      // 알림 처리
    });
  }
);

Kafka → Redis Pub/Sub → WebSocket 구조를 쓰는 이유

이슈 해결
멀티 인스턴스 환경에서 WebSocket 세션이 특정 서버에만 존재 Redis Pub/Sub으로 전체 서버에 브로드캐스트 → 해당 유저가 연결된 서버가 WebSocket 전송
알림 발생 시점과 WebSocket 전송을 분리하고 싶음 Kafka notification 토픽으로 비동기 처리 → 발신 측이 블로킹되지 않음
알림 이력이 필요 sendInternal()에서 DB 저장 후 Kafka 발행 — 이력과 실시간 푸시 동시 처리

미완성 항목

NotificationKafkaConsumer의 멱등성 보장 미구현 — 같은 notificationId로 중복 메시지가 들어올 경우 Redis Pub/Sub가 중복 발행될 수 있음. DB 조회로 존재 여부만 확인하고 있으나 중복 전송 방지 처리 필요.
